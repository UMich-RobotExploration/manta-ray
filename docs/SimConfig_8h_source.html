178/// @brief Cross-field validation on a loaded SimConfig.
179/// @details Emits warnings for parameter combinations that will produce
180/// silently wrong-looking diagnostics downstream. The sim itself remains
181/// valid to run; the warnings surface issues the user might otherwise
182/// spend time chasing as bugs.
183inline void validateSimConfig(const SimConfig &c) {
184 const double poseDtSec =
185 (c.sensors.odomFreqHz > 0.0) ? 1.0 / c.sensors.odomFreqHz : 0.0;
186 const double pingIntervalSec = c.pingIntervalMin * 60.0;
187 if (poseDtSec > 0.0 && pingIntervalSec > 0.0) {
188 const double ratio = pingIntervalSec / poseDtSec;
189 const double residual = std::abs(ratio - std::round(ratio));
190 if (residual > 1e-6) {
191 // Stale-pose: ping_interval is not an integer multiple of pose dt, so
192 // every range measurement references a pose vertex up to pose_dt/2
193 // seconds away from the ping timestamp. Downstream tools that compare
194 // EDGE_RANGE against VERTEX_SE3:QUAT positions (e.g.
195 // make_all_ranges_perfect) will report a residual that reflects this
196 // time misalignment rather than any acoustic effect.
197 const double suggestedSlowerSeconds = std::ceil(ratio) * poseDtSec;
198 const double suggestedFasterSeconds = std::floor(ratio) * poseDtSec;
199 SPDLOG_WARN(
200 "Stale-pose: ping_interval ({:.1f} s) is not a whole multiple of "
201 "pose dt ({:.1f} s = 1/odom_freq_hz). Range factors will reference "
202 "poses up to {:.1f} s stale, producing a residual in downstream "
203 "diagnostics that reflects time misalignment rather than acoustics."
204 " Set ping_interval_min to a multiple of {:.3f} sec "
205 "(nearest slower ping: {:.3f} sec, nearest faster ping: {:.3f} sec).",
206 pingIntervalSec, poseDtSec, 0.5 * poseDtSec, poseDtSec,
207 suggestedSlowerSeconds, suggestedFasterSeconds);
208 if (auto lg = spdlog::default_logger()) {
209 lg->flush();
210 }
211 }
212 }
213}

215inline SimConfig loadSimConfig(const std::string &path) {
216 std::ifstream f(path);
217 if (!f.is_open()) {
218 throw std::runtime_error("Cannot open sim config file: " + path);
219 }
220 auto j = json::parse(f);
221 auto cfg = j.get<SimConfig>();
222 validateSimConfig(cfg);
223 return cfg;
224}
